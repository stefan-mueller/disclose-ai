AI Disclosure for University Assignments

This tool generates a plain-text AI Disclosure Declaration that summarises which generative AI tools you used, and how, so you can paste it into your assignment appendix. None of the information or details you enter is stored or transmitted anywhere — it all stays in your browser.

Built by Stefan Müller and Claude Code. This idea is inspired by the GAIDeT Declaration Generator (Suchikova et al. 2026), adapted for university assignments.

Tick every task you delegated to a generative AI tool. For each ticked task, add one sentence describing what you asked the AI to do and how you used its output. When you are finished, copy the declaration at the end of this page and make manual adjustments where needed.

function el(tag, props = {}, children = []) {
  const node = document.createElement(tag);
  for (const [k, v] of Object.entries(props)) {
    if (v === undefined || v === null) continue;
    if (k.startsWith("on") && typeof v === "function") {
      node.addEventListener(k.slice(2), v);
    } else if (k === "checked" || k === "value") {
      node[k] = v;
    } else if (k === "class") {
      node.className = v;
    } else {
      node.setAttribute(k, v);
    }
  }
  for (const child of [].concat(children)) {
    if (child === null || child === undefined) continue;
    node.append(child instanceof Node ? child : document.createTextNode(child));
  }
  return node;
}

function debounce(fn, delay = 500) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}

function slowScrollTo(target, duration = 1200) {
  const startY = window.scrollY;
  const targetY = startY + target.getBoundingClientRect().top - 20;
  const diff = targetY - startY;
  const start = performance.now();
  function step(now) {
    const t = Math.min((now - start) / duration, 1);
    const eased = t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
    window.scrollTo(0, startY + diff * eased);
    if (t < 1) requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}

function preserveFocus(build) {
  const active = document.activeElement;
  const field = active?.dataset?.field;
  const selStart = active?.selectionStart ?? null;
  const selEnd = active?.selectionEnd ?? null;
  const node = build();
  if (field) {
    const refocus = () => {
      const next = node.querySelector(`[data-field="${CSS.escape(field)}"]`);
      if (!next) return;
      if (!next.isConnected) {
        requestAnimationFrame(refocus);
        return;
      }
      next.focus({ preventScroll: true });
      if (selStart !== null && typeof next.setSelectionRange === "function") {
        next.setSelectionRange(selStart, selEnd);
      }
    };
    queueMicrotask(refocus);
  }
  return node;
}

Detailed task breakdown

declarationText = {
  const entries = Object.values(selections);

  const overallToolNames = Object.keys(overallTools).map(tool => {
    const version = (overallToolVersions[tool] ?? "").trim();
    return version ? `${tool} (${version})` : tool;
  });
  overallToolNames.push(...Object.values(overallOtherTools).map(v => v.trim()).filter(Boolean));
  const overallToolsPart = overallToolNames.length > 0 ? overallToolNames.join(", ") : "[no tools ticked]";

  const heading = `AI Disclosure Declaration\n\nAI tools used: ${overallToolsPart}\n`;

  const signedBy = studentName.trim() || "[ADD NAME]";
  const closing = `\n\nResponsibility for the final manuscript lies entirely with the author.\nGenerative AI tools are not listed as authors and do not bear responsibility for the final outcomes.\nDeclaration submitted by: ${signedBy}`;

  if (entries.length === 0) {
    return `${heading}\nNo tasks have been ticked yet.${closing}`;
  }

  const lines = entries.map(e => {
    const labelPart = (e.customLabel && e.customLabel.trim()) ? e.customLabel.trim() : e.label;
    const itemToolNames = Object.keys(e.tools ?? {});
    const itemToolsPart = itemToolNames.length > 0 ? itemToolNames.join(", ") : "[no tool specified]";
    const toolLabel = itemToolNames.length === 1 ? "Tool" : "Tools";
    const sentencePart = e.sentence.trim() || "[please add a short explainer]";
    const promptsPart = providePrompts && e.prompts && e.prompts.trim() ? `\n    Prompts: ${e.prompts.trim()}` : "";
    return `- ${labelPart}\n    ${toolLabel}: ${itemToolsPart}\n    Summary: ${sentencePart}${promptsPart}`;
  });

  return `${heading}\nThe following generative AI tasks were delegated as part of this assignment:\n\n${lines.join("\n\n")}${closing}`;
}

{
  const noTools = Object.keys(overallTools).length === 0
    && Object.values(overallOtherTools).every(v => !v.trim());
  const noTasks = Object.values(selections).length === 0;
  const taskMissingTool = Object.values(selections).some(sel =>
    Object.keys(sel.tools ?? {}).length === 0
  );
  const taskMissingLabel = Object.values(selections).some(sel =>
    sel.openEnded && !(sel.customLabel && sel.customLabel.trim())
  );
  const incomplete = noTools || noTasks || taskMissingTool || taskMissingLabel;

  const warningText = (noTools || noTasks)
    ? "If you have used AI tool(s) in your assignment, please select at least one tool above and tick at least one task before copying the declaration."
    : taskMissingTool
    ? "Please select at least one tool for every ticked task before copying the declaration."
    : "Please add a short description for every “Not listed above” task before copying the declaration.";

  const warning = (incomplete && copyWarning)
    ? el("div", { class: "gaidet-warning" }, warningText)
    : null;

  const successText = copySuccess === "download"
    ? "Declaration downloaded as a text file — you can open it, paste it into your assignment, and make manual edits if needed."
    : "Declaration copied to clipboard — you can paste it into your assignment and make manual edits if needed.";

  const success = (!incomplete && copySuccess)
    ? el("div", { class: "gaidet-success" }, successText)
    : null;

  function guarded(action) {
    return () => {
      if (incomplete) {
        mutable copyWarning = true;
        mutable copySuccess = false;
        return;
      }
      mutable copyWarning = false;
      action();
    };
  }

  return el("div", { class: "gaidet-output" }, [
    el("h3", {}, "Generated declaration"),
    warning,
    success,
    el("pre", { id: "gaidet-output-text" }, [
      el("strong", {}, "AI Disclosure Declaration"),
      declarationText.replace(/^AI Disclosure Declaration/, "")
    ]),
    el("div", { class: "gaidet-output-actions" }, [
      el("button", {
        class: "gaidet-btn-primary",
        onclick: guarded(() => {
          navigator.clipboard.writeText(declarationText);
          mutable copySuccess = "copy";
        })
      }, "Copy to clipboard"),
      el("button", {
        class: "gaidet-btn-secondary",

          const a = document.createElement("a");
          a.href = url;
          a.download = "ai-disclosure-declaration.txt";
          a.click();
          URL.revokeObjectURL(url);
          mutable copySuccess = "download";
        })
      }, "Download as file")
    ])
